--- stand-in full-screen program (conmouse.ps1) ---
# Stand-in Windows full-screen program: asks the console for mouse input and counts wheel events.
Add-Type -TypeDefinition @"
using System;
using System.Runtime.InteropServices;
public static class Con {
  [StructLayout(LayoutKind.Explicit)] public struct REC {
    [FieldOffset(0)] public ushort Type;
    [FieldOffset(4)] public int KeyDown;
    [FieldOffset(10)] public ushort VKey;
    [FieldOffset(14)] public char Ch;
    [FieldOffset(4)] public short X;
    [FieldOffset(6)] public short Y;
    [FieldOffset(8)] public uint Buttons;
    [FieldOffset(16)] public uint Flags;
  }
  [DllImport("kernel32.dll")] public static extern IntPtr GetStdHandle(int n);
  [DllImport("kernel32.dll")] public static extern bool GetConsoleMode(IntPtr h, out uint m);
  [DllImport("kernel32.dll")] public static extern bool SetConsoleMode(IntPtr h, uint m);
  [DllImport("kernel32.dll", CharSet=CharSet.Unicode)] public static extern bool ReadConsoleInput(IntPtr h, [Out] REC[] r, uint n, out uint got);
}
"@
$h = [Con]::GetStdHandle(-10)
$old = 0; [void][Con]::GetConsoleMode($h, [ref]$old)
[void][Con]::SetConsoleMode($h, 0x0010 -bor 0x0080 -bor 0x0008)
$up = 0; $down = 0
[Console]::Clear()
function Draw { [Console]::SetCursorPosition(0,0); [Console]::Write("CONSOLE-MOUSE wheel up=$up down=$down (q quits)    ") }
Draw
$buf = New-Object Con+REC[] 16
while ($true) {
  $got = 0; [void][Con]::ReadConsoleInput($h, $buf, 16, [ref]$got)
  $quit = $false
  for ($i = 0; $i -lt $got; $i++) {
    $r = $buf[$i]
    if ($r.Type -eq 2 -and $r.Flags -eq 4) { if (($r.Buttons -band 0x80000000) -eq 0) { $up++ } else { $down++ } }
    if ($r.Type -eq 1 -and $r.KeyDown -ne 0 -and $r.Ch -eq 'q') { $quit = $true }
  }
  Draw
  if ($quit) { break }
}
[void][Con]::SetConsoleMode($h, $old)
[Console]::Clear()
