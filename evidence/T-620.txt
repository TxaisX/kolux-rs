WHY ONLY ONE SPOT MOVED THE WINDOW
Tauri 2.12.1's drag script (crate tauri-2.12.1, src/window/scripts/drag.js, isDragRegion) walks up
from the element that was pressed:
  bare or "true" data-tauri-drag-region -> drags only when the press lands on that element itself
  "deep"                               -> drags from anywhere inside it
  "false"                              -> never drags from here
  a button, link, input or menu item without the attribute -> never drags
The bar (src/chrome/TopBar.tsx) carried the bare attribute, but its left group fills the whole left
third and the right group the right, so only the strips between them were the header element itself.

WHAT CHANGED
1. src/chrome/TopBar.tsx: the header is data-tauri-drag-region="deep". The switcher's frame (nav) and
   the View menu's backdrop and list opt out ("false"): a press that just misses a segment does not
   move the window, and a click on a menu's backdrop still closes the menu. The app name's own bare
   attribute is gone ("deep" covers it). Header comment updated.
2. src/switcher/AccountSwitcher.tsx: "false" on the account menu's backdrop and list (the switcher is
   drawn inside the bar).
3. src/editor/TableControls.tsx: the table's cell menu closes on pointerdown instead of mousedown (new
   watchMenuClose) and the menu stops its own pointerdown. drag.js stops the mousedown of a press on a
   drag region at the document, so once the whole bar was one, a press on the bar no longer reached the
   menu's window listener (found by the fresh review).
4. src/chrome/topBarDrag.test.ts: applies drag.js's rules to every element the bar draws, modern and
   Classic, and with the View and account menus open: anything a person uses (a control, a menu, an
   element with a React press or click handler, the switcher's frame) never drags, and everything else
   does. Handlers are read from React's props, so a clickable span or a new pop-up added to the bar
   later fails the test without the test changing.
5. src/editor/TableControls.test.ts (new): pointerdown and Escape close the cell menu; nothing listens
   once it is closed.

----- drag-proof.cjs -----
// T-620 proof: load the real TopBar (Vite dev server) with Tauri 2.12.1's own drag.js and a recording
// IPC stand-in, press across the whole bar, and report where a press starts a window drag.
// What counts as "something a person uses" is decided independently of drag.js: a form control, an element
// with a React press or click handler (read from React's props on the node), a menu, or the switcher's
// frame. Then every control is clicked for real, on a fresh page, to show it still does its job.
// usage: node drag-proof.cjs <url> <label>  -> prints a summary and writes <label>.json beside this file
const { chromium } = require("playwright");
const fs = require("fs");
const path = require("path");
const [url, label] = process.argv.slice(2);

// Runs in the page: is this element (or one around it, inside the bar) something a person uses?
const CLASSIFY = () => {
  const HANDLERS = ["onClick", "onMouseDown", "onMouseUp", "onPointerDown", "onPointerUp", "onDoubleClick", "onContextMenu"];
  const FORM = new Set(["A", "BUTTON", "INPUT", "SELECT", "TEXTAREA", "LABEL", "SUMMARY"]);
  const props = (n) => { const k = Object.keys(n).find((key) => key.startsWith("__reactProps$")); return k ? n[k] : {}; };
  const usable = (n) => FORM.has(n.tagName) || n.getAttribute("role") === "menu" || n.tagName === "NAV" || HANDLERS.some((h) => typeof props(n)[h] === "function");
  window.__usedBy = (el) => { for (let n = el; n && n !== document.body; n = n.parentElement) if (usable(n)) return n; return null; };
  window.__name = (n) => (n ? n.getAttribute("title") || n.getAttribute("aria-label") || (typeof n.className === "string" && n.className) || n.tagName.toLowerCase() : "nothing");
};

async function open(browser, path) {
  const page = await browser.newPage({ viewport: { width: 1280, height: 300 } });
  const errors = [];
  page.on("pageerror", (e) => errors.push(String(e)));
  await page.goto(path);
  await page.waitForSelector("header");
  await page.waitForTimeout(400);
  await page.evaluate(CLASSIFY);
  return { page, errors };
}

(async () => {
  const browser = await chromium.launch();
  const { page, errors } = await open(browser, url);
  // While scanning, swallow clicks so a press on a control does not open a menu and change the page;
  // count which presses reach the window as mousedown and as pointerdown (the table menu listens there).
  await page.evaluate(() => {
    window.__scan = true;
    window.__win = { mousedown: 0, pointerdown: 0 };
    window.addEventListener("click", (e) => { if (window.__scan) { e.stopPropagation(); e.preventDefault(); } }, true);
    window.addEventListener("mousedown", () => window.__win.mousedown++);
    window.addEventListener("pointerdown", () => window.__win.pointerdown++);
  });
  const bar = await page.$eval("header", (h) => { const r = h.getBoundingClientRect(); return { x: r.x, y: r.y, w: r.width, h: r.height, attr: h.getAttribute("data-tauri-drag-region") }; });
  const probe = async (x, y) => {
    const hit = await page.evaluate(([x, y]) => {
      const el = document.elementFromPoint(x, y);
      const used = window.__usedBy(el);
      window.__calls.length = 0;
      window.__win.mousedown = window.__win.pointerdown = 0;
      return { control: !!used, what: window.__name(used || el) };
    }, [x, y]);
    await page.mouse.move(x, y);
    await page.mouse.down();
    await page.mouse.up();
    const after = await page.evaluate(() => ({ calls: window.__calls.slice(), win: { ...window.__win } }));
    return { x, y, ...hit, drags: after.calls.includes("plugin:window|start_dragging"), win: after.win };
  };
  const rows = [];
  for (const dy of [3, 18, 32]) for (let x = 2; x < bar.x + bar.w; x += 6) rows.push(await probe(x, bar.y + dy));
  const empty = rows.filter((r) => !r.control), ctl = rows.filter((r) => r.control);
  const group = (list) => [...list.reduce((m, r) => m.set(r.what, (m.get(r.what) || 0) + 1), new Map())].map(([k, n]) => `${k} x${n}`);
  const out = {
    label, bar,
    points: rows.length,
    emptyPoints: empty.length, emptyThatDrag: empty.filter((r) => r.drags).length,
    controlPoints: ctl.length, controlsThatDrag: ctl.filter((r) => r.drags).length,
    emptyNotDragging: group(empty.filter((r) => !r.drags)),
    controlsDragging: group(ctl.filter((r) => r.drags)),
    controlsPressed: group(ctl),
    emptyPressesReachingWindow: { mousedown: empty.filter((r) => r.win.mousedown).length, pointerdown: empty.filter((r) => r.win.pointerdown).length, of: empty.length },
  };
  // Double-click on an empty spot of the bar: the second press sends Tauri's maximize toggle.
  const spot = empty.find((r) => r.drags) || empty[0];
  await page.evaluate(() => (window.__calls.length = 0));
  await page.mouse.click(spot.x, spot.y, { clickCount: 2 });
  out.doubleClick = { at: [spot.x, spot.y, spot.what], calls: (await page.evaluate(() => window.__calls.slice())).filter((c) => c.startsWith("plugin:window|")) };
  // The View menu and the account menu: open each for real, then press its backdrop and its title.
  await page.evaluate(() => (window.__scan = false));
  const pressNoUp = async (sel, x, y) => {
    await page.evaluate(() => (window.__calls.length = 0));
    if (sel) [x, y] = await page.$eval(sel, (n) => { const r = n.getBoundingClientRect(); return [r.x + 6, r.y + r.height / 2]; });
    await page.mouse.move(x, y);
    await page.mouse.down();
    const drags = (await page.evaluate(() => window.__calls.slice())).includes("plugin:window|start_dragging");
    await page.mouse.up();
    return drags;
  };
  out.menus = {};
  for (const [button, menuLabel] of [['button[title="Menu"]', "View"], ['button[title="Choose the account new panes use"]', "Accounts for new panes"]]) {
    if (!(await page.$(button))) { out.menus[menuLabel] = "button not drawn"; continue; }
    await page.click(button);
    await page.waitForSelector(`[role="menu"][aria-label="${menuLabel}"]`);
    const m = { backdropAttr: await page.$eval(".menu-backdrop", (n) => n.getAttribute("data-tauri-drag-region")), listAttr: await page.$eval(`[role="menu"][aria-label="${menuLabel}"]`, (n) => n.getAttribute("data-tauri-drag-region")) };
    m.pressOnTitleDrags = await pressNoUp(`[role="menu"][aria-label="${menuLabel}"] .menu-title`);
    m.pressOnBackdropDrags = await pressNoUp(null, 640, 200);
    m.backdropClickClosedIt = (await page.$(`[role="menu"][aria-label="${menuLabel}"]`)) === null;
    // Without the "false" opt-out the same backdrop press would move the window instead of closing the menu.
    await page.click(button);
    await page.waitForSelector(`[role="menu"][aria-label="${menuLabel}"]`);
    await page.$eval(".menu-backdrop", (n) => n.removeAttribute("data-tauri-drag-region"));
    m.withoutOptOutBackdropPressDrags = await pressNoUp(null, 640, 200);
    out.menus[menuLabel] = m;
    await page.reload();
    await page.waitForSelector("header");
    await page.waitForTimeout(300);
  }
  await page.close();
  // Every control, clicked for real on a fresh page: it does its job (a handler, a command, a menu or an
  // event) and never moves the window.
  const probePage = await open(browser, url);
  const controls = await probePage.page.evaluate(() => [...document.querySelectorAll("header button")].filter((b) => b.getClientRects().length > 0 && !b.disabled).map((b) => window.__name(b)));
  await probePage.page.close();
  out.controls = [];
  for (let i = 0; i < controls.length; i++) {
    const { page: p } = await open(browser, url);
    await p.evaluate(() => { window.__calls.length = 0; window.__props.length = 0; window.__events.length = 0; });
    const box = await p.evaluate((i) => { const b = [...document.querySelectorAll("header button")].filter((b) => b.getClientRects().length > 0 && !b.disabled)[i]; const r = b.getBoundingClientRect(); return [r.x + r.width / 2, r.y + r.height / 2]; }, i);
    await p.mouse.click(box[0], box[1]);
    await p.waitForTimeout(150);
    const did = await p.evaluate(() => ({
      props: window.__props.slice(),
      calls: window.__calls.filter((c) => !["plugin:window|is_maximized", "plugin:event|listen", "claude_accounts", "codex_accounts"].includes(c) && !c.includes("usage")),
      events: window.__events.filter((e) => !e.startsWith("vite")),
      menu: [...document.querySelectorAll('[role="menu"]')].map((m) => m.getAttribute("aria-label")),
    }));
    out.controls.push({ control: controls[i], drags: did.calls.includes("plugin:window|start_dragging"), did: [...did.props, ...did.calls, ...did.events.map((e) => "event " + e), ...did.menu.map((m) => "menu " + m)].join(", ") || "NOTHING" });
    await p.close();
  }
  out.errors = errors.filter((e) => !e.includes("not in the harness"));
  fs.writeFileSync(path.join(__dirname, `${label}.json`), JSON.stringify({ ...out, rows }, null, 1));
  console.log(JSON.stringify(out, null, 1));
  await browser.close();
})().catch((e) => { console.error(e); process.exit(1); });
