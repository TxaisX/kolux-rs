Proved in the Linux cloud sandbox (Chromium + Playwright, Vite dev server, node 22). Nothing was run
on Txais's PC. CI is set aside (Txais, 2026-10-30).

WHY ONLY ONE SPOT MOVED THE WINDOW
Tauri 2.12.1's drag script (crate tauri-2.12.1, src/window/scripts/drag.js, isDragRegion) walks up
from the element that was pressed:
  bare or "true" data-tauri-drag-region -> drags only when the press lands on that element itself
  "deep"                               -> drags from anywhere inside it
  "false"                              -> never drags from here
  a button, link, input or menu item without the attribute -> never drags
The bar (src/chrome/TopBar.tsx) carried the bare attribute, but its left group fills the whole left
third and the right group sits on the right, so only the gaps around the centred switcher were the
header element itself.

WHAT CHANGED
1. src/chrome/TopBar.tsx: the header is data-tauri-drag-region="deep"; the View menu's backdrop and
   list (drawn inside the bar) carry "false", so a press on them closes or uses the menu instead of
   moving the window. Header comment updated.
2. src/switcher/AccountSwitcher.tsx: the same "false" on the account menu's backdrop and list (the
   switcher is mounted inside the bar).
3. src/chrome/topBarDrag.test.ts (new): the header is "deep"; the open View menu and the open
   account menu both carry "false".

  // A stand-in for Tauri's IPC: it records every command and answers the few the top bar reads.
  window.__calls = [];
  let nextId = 1;
  window.__TAURI_INTERNALS__ = {
    metadata: { currentWindow: { label: "main" }, currentWebview: { windowLabel: "main", label: "main" } },
    transformCallback: () => nextId++,
    unregisterCallback: () => {},
    convertFileSrc: (p) => p,
    invoke: (cmd) => {
      window.__calls.push(cmd);
      if (cmd === "plugin:window|is_maximized") return Promise.resolve(false);
      if (cmd === "plugin:event|listen") return Promise.resolve(nextId++);
      return Promise.reject("not in the harness: " + cmd);
    },
  };
  window.__TAURI_EVENT_PLUGIN_INTERNALS__ = { unregisterListener: () => {} };
</script>

----- drag-proof.cjs -----
// T-620 proof: load the real TopBar (Vite dev server) with Tauri 2.12.1's own drag.js and a recording
// IPC stand-in, press across the whole bar, and report where a press starts a window drag.
// usage: node drag-proof.cjs <url> <label>  -> prints a summary and writes <label>.json beside this file
const { chromium } = require("playwright");
const fs = require("fs");
const path = require("path");
const [url, label] = process.argv.slice(2);

const CONTROL = 'a,button,input,select,textarea,label,summary,[contenteditable]:not([contenteditable="false"]),[tabindex]:not([tabindex="-1"]),[role=button],[role=link],[role=menuitem],[role=tab],[role=checkbox],[role=radio],[role=switch],[role=option]';

(async () => {
  const browser = await chromium.launch();
  const page = await browser.newPage({ viewport: { width: 1280, height: 300 } });
  const errors = [];
  page.on("pageerror", (e) => errors.push(String(e)));
  await page.goto(url);
  await page.waitForSelector("header");
  await page.waitForTimeout(400);
  // While scanning, swallow clicks so a press on a control does not open a menu and change the page.
  await page.evaluate(() => {
    window.__scan = true;
    window.addEventListener("click", (e) => { if (window.__scan) { e.stopPropagation(); e.preventDefault(); } }, true);
  });
  const bar = await page.$eval("header", (h) => { const r = h.getBoundingClientRect(); return { x: r.x, y: r.y, w: r.width, h: r.height, attr: h.getAttribute("data-tauri-drag-region") }; });
  const probe = async (x, y) => {
    const hit = await page.evaluate(([x, y, CONTROL]) => {
      const el = document.elementFromPoint(x, y);
      const ctl = el && el.closest(CONTROL);
      const name = (n) => (n ? n.getAttribute("title") || n.getAttribute("aria-label") || (typeof n.className === "string" && n.className) || n.tagName.toLowerCase() : "nothing");
      window.__calls.length = 0;
      return { control: !!ctl, what: ctl ? name(ctl) : name(el) };
    }, [x, y, CONTROL]);
    await page.mouse.move(x, y);
    await page.mouse.down();
    await page.mouse.up();
    const calls = await page.evaluate(() => window.__calls.slice());
    return { x, y, ...hit, drags: calls.includes("plugin:window|start_dragging") };
  };
  const rows = [];
  for (const dy of [3, 18, 32]) for (let x = 2; x < bar.x + bar.w; x += 6) rows.push(await probe(x, bar.y + dy));
  const empty = rows.filter((r) => !r.control), ctl = rows.filter((r) => r.control);
  const group = (list) => [...list.reduce((m, r) => m.set(r.what, (m.get(r.what) || 0) + 1), new Map())].map(([k, n]) => `${k} x${n}`);
  const out = {
    label, bar,
    points: rows.length,
    emptyPoints: empty.length, emptyThatDrag: empty.filter((r) => r.drags).length,
    controlPoints: ctl.length, controlsThatDrag: ctl.filter((r) => r.drags).length,
    emptyNotDragging: group(empty.filter((r) => !r.drags)),
    controlsDragging: group(ctl.filter((r) => r.drags)),
  };
  // Double-click on an empty spot of the bar: the second press asks Tauri to maximize.
  const spot = empty.find((r) => r.drags) || empty[0];
  await page.evaluate(() => (window.__calls.length = 0));
  await page.mouse.click(spot.x, spot.y, { clickCount: 2 });
  out.doubleClick = { at: [spot.x, spot.y, spot.what], calls: (await page.evaluate(() => window.__calls.slice())).filter((c) => c.startsWith("plugin:window|")) };
  // The View menu: open it for real, then press its backdrop and its title.
  await page.evaluate(() => (window.__scan = false));
  await page.click('button[title="Menu"]');
  await page.waitForSelector('[role="menu"]');
  const pressNoUp = async (sel, x, y) => {
    await page.evaluate(() => (window.__calls.length = 0));
    if (sel) { const b = await page.$eval(sel, (n) => { const r = n.getBoundingClientRect(); return [r.x + 6, r.y + r.height / 2]; }); [x, y] = b; }
    await page.mouse.move(x, y);
    await page.mouse.down();
    const drags = (await page.evaluate(() => window.__calls.slice())).includes("plugin:window|start_dragging");
    await page.mouse.up();
    return drags;
  };
  out.menu = {
    backdropAttr: await page.$eval(".menu-backdrop", (n) => n.getAttribute("data-tauri-drag-region")),
    pressOnMenuTitleDrags: await pressNoUp(".menu-title"),
  };
  out.menu.pressOnBackdropDrags = await pressNoUp(null, 640, 200);
  out.menu.backdropClickClosedIt = (await page.$('[role="menu"]')) === null;
  // Without the "false" opt-out the same backdrop press would move the window instead of closing the menu.
  await page.click('button[title="Menu"]');
  await page.waitForSelector('[role="menu"]');
  await page.$eval(".menu-backdrop", (n) => n.removeAttribute("data-tauri-drag-region"));
  out.menu.withoutOptOutBackdropPressDrags = await pressNoUp(null, 640, 200);
  out.errors = errors.filter((e) => !e.includes("not in the harness"));
  fs.writeFileSync(path.join(__dirname, `${label}.json`), JSON.stringify({ ...out, rows }, null, 1));
  console.log(JSON.stringify(out, null, 1));
  await browser.close();
})().catch((e) => { console.error(e); process.exit(1); });
