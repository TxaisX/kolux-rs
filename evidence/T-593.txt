    const CLI: &str = if cfg!(debug_assertions) { "kolux-dev" } else { "kolux" };
    fn command_file_windows(dir: &std::path::Path) -> Result<(), String> {
        let name = CLI;
        if ["exe", "cmd"].iter().any(|e| dir.join(format!("{name}.{e}")).is_file()) {
            Ok(())
        } else {
            Err(format!("{} holds no {name}.exe or {name}.cmd, so there is nothing to register", dir.display()))
        }
    }
    fn main() {
        let own = format!("{}.cmd", CLI);
        for (label, files) in [("before", vec!["kolux.exe", "kolux.cmd"]), ("after", vec!["kolux.exe", "kolux.cmd", own.as_str()])] {
            let bin = std::env::temp_dir().join(format!("t593-model-{label}-{}", std::process::id())).join("bin");
            std::fs::create_dir_all(&bin).unwrap();
            for f in &files {
                std::fs::write(bin.join(f), "").unwrap();
            }
            println!("CLI = {CLI}; {label}: bin holds {files:?} -> command_file = {:?}", command_file_windows(&bin));
            std::fs::remove_dir_all(bin.parent().unwrap()).unwrap();
        }
    }
