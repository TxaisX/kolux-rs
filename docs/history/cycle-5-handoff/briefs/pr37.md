pub fn workspace_files(workspace: &Path) -> Vec<PathBuf> {
    let mut out = Vec::new();
    let dir = workspace.join(WORKSPACE_DIR);
    // The folder itself must be a real directory inside the workspace: read_dir would follow a
    // symlink or a Windows junction (symlink_metadata reports both as symlinks, so is_dir() is
    // false for them), and canonicalize catches any other way out.

        }
    }
    out.sort();
    out
}
